-- =====================================================================
--  Post-operative remote patient monitoring
--  Neon PostgreSQL schema  --  v0.1
--
--  Two schemas:
--    meta  : entity definitions and clinical rules (slow-changing)
--    txn   : patient data, check-ins, evaluations, alerts (fast-growing)
--
--  Run as a single transaction so a partial apply cannot happen.
-- =====================================================================

begin;

create schema if not exists meta;
create schema if not exists txn;

-- gen_random_uuid() is built into PostgreSQL 13+, so no pgcrypto needed.
-- Neon runs 16/17, so this is safe.

-- ---------------------------------------------------------------------
-- shared: updated_at trigger
-- ---------------------------------------------------------------------
create or replace function meta.set_updated_at() returns trigger
language plpgsql as $$
begin
    new.updated_at = now();
    return new;
end;
$$;


-- =====================================================================
--  META
-- =====================================================================

-- ---------------------------------------------------------------------
-- hospital
-- ---------------------------------------------------------------------
create table meta.hospital (
    hospital_id       integer generated always as identity primary key,
    hospital_code     text        not null unique,          -- short stable key, e.g. 'CBE-ORTHO-01'
    hospital_name     text        not null,
    city              text,
    timezone          text        not null default 'Asia/Kolkata',
    ack_window_secs   integer     not null default 900,     -- escalation timer, per hospital
    is_active         boolean     not null default true,
    created_at        timestamptz not null default now(),
    updated_at        timestamptz not null default now(),
    constraint hospital_ack_window_sane
        check (ack_window_secs between 60 and 7200)
);

create trigger hospital_touch before update on meta.hospital
    for each row execute function meta.set_updated_at();


-- ---------------------------------------------------------------------
-- surgical_procedure
--   NOTE: "procedure" is a RESERVED keyword in PostgreSQL. A table named
--   procedure must be double-quoted at every single reference forever.
--   Renamed to surgical_procedure.
-- ---------------------------------------------------------------------
create table meta.surgical_procedure (
    procedure_id      integer generated always as identity primary key,
    procedure_code    text        not null unique,          -- 'TKR', 'THR'
    procedure_name    text        not null,
    procedure_desc    text,
    observation_days  smallint    not null,                 -- was duration_of_observation
    is_active         boolean     not null default true,
    created_at        timestamptz not null default now(),
    updated_at        timestamptz not null default now(),
    constraint procedure_observation_days_sane
        check (observation_days between 1 and 365)
);

create trigger procedure_touch before update on meta.surgical_procedure
    for each row execute function meta.set_updated_at();


-- ---------------------------------------------------------------------
-- hospital_procedure   (which hospital offers which procedure)
--   CHANGED: dropped hospital_name. It duplicated meta.hospital and would
--   drift the first time a hospital is renamed. Join for it instead.
-- ---------------------------------------------------------------------
create table meta.hospital_procedure (
    hospital_id   integer not null references meta.hospital(hospital_id)            on delete cascade,
    procedure_id  integer not null references meta.surgical_procedure(procedure_id) on delete restrict,
    enabled_on    date    not null default current_date,
    is_active     boolean not null default true,
    primary key (hospital_id, procedure_id)
);


-- ---------------------------------------------------------------------
-- parameter   (the registry of things that can be tracked)
--   CHANGED: valid_range split into typed min/max columns so the database
--   can actually enforce it. valid_values moved to parameter_option below.
-- ---------------------------------------------------------------------
create table meta.parameter (
    parameter_id    integer generated always as identity primary key,
    parameter_code  text        not null unique,     -- 'temp_c', 'calf_pain', 'knee_bend'
    parameter_name  text        not null,            -- 'Temperature'
    parameter_desc  text,
    parameter_type  text        not null,            -- numeric | boolean | ordinal
    unit            text,                            -- 'C', 'bpm', '%', null for boolean
    value_min       numeric(10,2),                   -- numeric types only
    value_max       numeric(10,2),
    decimal_places  smallint    not null default 0,
    concerning_when_true boolean,                    -- booleans: which answer is the worrying one
    is_active       boolean     not null default true,
    created_at      timestamptz not null default now(),

    constraint parameter_type_known
        check (parameter_type in ('numeric','boolean','ordinal')),
    constraint parameter_range_ordered
        check (value_min is null or value_max is null or value_min <= value_max),
    constraint parameter_numeric_needs_range
        check (parameter_type <> 'numeric'
               or (value_min is not null and value_max is not null)),
    constraint parameter_boolean_needs_polarity
        check (parameter_type <> 'boolean' or concerning_when_true is not null)
);


-- ---------------------------------------------------------------------
-- parameter_option   (valid_values for ordinal parameters)
--   ordinal_rank is what makes 'walker < crutches < cane < none' comparable,
--   which a comma-separated valid_values string could never do.
-- ---------------------------------------------------------------------
create table meta.parameter_option (
    parameter_id  integer  not null references meta.parameter(parameter_id) on delete cascade,
    option_code   text     not null,          -- 'chair', 'car', 'bike'
    option_label  text     not null,          -- 'Sit in a dining chair'
    ordinal_rank  smallint not null,          -- 1, 2, 3 — higher is better progress
    primary key (parameter_id, option_code),
    unique (parameter_id, ordinal_rank)
);


-- ---------------------------------------------------------------------
-- ruleset   (NEW — the version wrapper around scoring)
--   Weights and bands change when a surgeon revises them. If they hang
--   straight off procedure_id, editing a weight silently rewrites the
--   reasoning behind every past alert. Versioning makes decisions
--   reconstructable, which is the first thing asked for when a clinician
--   disputes an alert.
-- ---------------------------------------------------------------------
create table meta.ruleset (
    ruleset_id               integer generated always as identity primary key,
    procedure_id             integer     not null references meta.surgical_procedure(procedure_id),
    version                  text        not null,              -- '1.3.0'
    review_at                smallint    not null,              -- weighted-total cutoffs
    critical_at              smallint    not null,
    max_points_per_parameter smallint    not null default 6,
    status                   text        not null default 'draft',
    approved_by              text,
    approved_at              timestamptz,
    notes                    text,
    created_at               timestamptz not null default now(),

    unique (procedure_id, version),
    constraint ruleset_status_known
        check (status in ('draft','approved','retired')),
    constraint ruleset_tiers_ordered
        check (critical_at > review_at and review_at > 0),
    constraint ruleset_approved_has_approver
        check (status <> 'approved'
               or (approved_by is not null and approved_at is not null))
);

-- at most one approved ruleset per procedure at any time
create unique index ruleset_one_approved_per_procedure
    on meta.ruleset (procedure_id)
    where status = 'approved';


-- ---------------------------------------------------------------------
-- procedure_parameter   (was procedure_param_map)
--   CHANGED 1: keyed on ruleset_id, not procedure_id (see above).
--   CHANGED 2: added track_to_day. You had a start but no end, and the
--              spec needs windows that close — oxygen days 0-14,
--              drainage days 5-28.
--   CHANGED 3: scoring config stored as jsonb, not yaml text. Authors can
--              still write YAML; the loader converts. jsonb is queryable,
--              validatable and indexable. A text blob is none of those.
-- ---------------------------------------------------------------------
create table meta.procedure_parameter (
    ruleset_id       integer  not null references meta.ruleset(ruleset_id)      on delete cascade,
    parameter_id     integer  not null references meta.parameter(parameter_id)  on delete restrict,
    display_order    smallint not null,
    display_section  text     not null default 'readings',   -- readings | checks | progress
    track_from_day   smallint not null default 0,
    track_to_day     smallint not null,
    is_mandatory     boolean  not null default false,
    parameter_weight smallint not null,
    scoring_config   jsonb    not null,        -- bands / bands_by_day / binary_score / detector
    question_text    text,                     -- patient-facing wording
    nurse_script     text,                     -- what to ask when this rule fires

    primary key (ruleset_id, parameter_id),
    unique (ruleset_id, display_order),
    constraint ppm_window_ordered   check (track_to_day >= track_from_day),
    constraint ppm_window_sane      check (track_from_day >= 0 and track_to_day <= 365),
    constraint ppm_weight_sane      check (parameter_weight between 0 and 10),
    constraint ppm_section_known    check (display_section in ('readings','checks','progress')),
    constraint ppm_scoring_is_object check (jsonb_typeof(scoring_config) = 'object')
);

create index procedure_parameter_by_param on meta.procedure_parameter (parameter_id);


-- =====================================================================
--  TRANSACTIONAL
-- =====================================================================

-- ---------------------------------------------------------------------
-- patient   (the person — stable across surgeries)
-- ---------------------------------------------------------------------
create table txn.patient (
    patient_id    uuid        primary key default gen_random_uuid(),
    hospital_id   integer     not null references meta.hospital(hospital_id),
    mrn           text,                                   -- hospital record number, typed in manually
    patient_name  text        not null,
    phone_e164    text        not null,
    language_code text        not null default 'en',
    date_of_birth date,
    created_at    timestamptz not null default now(),
    updated_at    timestamptz not null default now(),

    unique (hospital_id, mrn),
    constraint patient_language_known check (language_code in ('en','ta','hi')),
    constraint patient_phone_e164     check (phone_e164 ~ '^\+[1-9][0-9]{7,14}$')
);

create trigger patient_touch before update on txn.patient
    for each row execute function meta.set_updated_at();


-- ---------------------------------------------------------------------
-- patient_episode   (NEW — one monitored recovery)
--   This is the table your design most needed. patient_check_in was
--   carrying patient_id + procedure_id + hospital_id on every single row,
--   which is the signal that an episode entity is missing. It also gives
--   discharge_date a home — without it, day_after_discharge cannot be
--   validated or recomputed — and lets a patient have a second surgery
--   later without corrupting the first episode's history.
-- ---------------------------------------------------------------------
create table txn.patient_episode (
    episode_id         uuid        primary key default gen_random_uuid(),
    patient_id         uuid        not null references txn.patient(patient_id)             on delete restrict,
    hospital_id        integer     not null references meta.hospital(hospital_id),
    procedure_id       integer     not null references meta.surgical_procedure(procedure_id),
    ruleset_id         integer     not null references meta.ruleset(ruleset_id),
    surgery_date       date        not null,
    discharge_date     date        not null,
    monitoring_ends_on date        not null,
    consent_version    text        not null,
    consent_at         timestamptz not null,
    enrolled_by        text,
    status             text        not null default 'active',
    created_at         timestamptz not null default now(),

    constraint episode_status_known    check (status in ('active','completed','withdrawn')),
    constraint episode_dates_ordered   check (discharge_date >= surgery_date),
    constraint episode_end_after_disch check (monitoring_ends_on > discharge_date)
);

-- a patient cannot be in two active monitoring programmes at once
create unique index episode_one_active_per_patient
    on txn.patient_episode (patient_id)
    where status = 'active';

create index episode_active_by_hospital
    on txn.patient_episode (hospital_id, status);


-- ---------------------------------------------------------------------
-- episode_baseline   (discharge readings — the comparison anchor)
-- ---------------------------------------------------------------------
create table txn.episode_baseline (
    episode_id   uuid     not null references txn.patient_episode(episode_id) on delete cascade,
    parameter_id integer  not null references meta.parameter(parameter_id),
    value_num    numeric(10,2),
    value_bool   boolean,
    value_code   text,
    recorded_at  timestamptz not null default now(),
    primary key (episode_id, parameter_id),
    constraint baseline_exactly_one_value check (
        (value_num  is not null)::int
      + (value_bool is not null)::int
      + (value_code is not null)::int = 1)
);


-- ---------------------------------------------------------------------
-- patient_check_in   (the submission header)
--   CHANGED: split from your single table. The original mixed two models
--   at once — fixed columns (temperature, pulse, oxygen, pain_score) AND
--   an EAV pair (parameter_id, parameter_value). That means one row per
--   parameter while ALSO repeating the four vitals on every one of those
--   rows. It also defeats the parameter registry: adding a parameter
--   would need a schema change, which is the thing meta.parameter exists
--   to avoid. Header here, values in the child table below.
-- ---------------------------------------------------------------------
create table txn.patient_check_in (
    check_in_id         uuid        primary key default gen_random_uuid(),
    episode_id          uuid        not null references txn.patient_episode(episode_id) on delete cascade,
    day_after_discharge smallint    not null,
    submitted_at        timestamptz not null default now(),
    captured_at         timestamptz,                    -- set on device; survives the offline queue
    source              text        not null default 'patient',
    idempotency_key     text,                           -- stops offline replay creating duplicates
    photo_key           text,                           -- object storage key, never a URL
    is_superseded       boolean     not null default false,

    constraint checkin_day_sane   check (day_after_discharge between 0 and 365),
    constraint checkin_source_known
        check (source in ('patient','proxy','nurse','synthetic'))
);

create unique index check_in_idempotency
    on txn.patient_check_in (episode_id, idempotency_key)
    where idempotency_key is not null;

-- drives "latest submission for this day"
create index check_in_by_day
    on txn.patient_check_in (episode_id, day_after_discharge, submitted_at desc);


-- ---------------------------------------------------------------------
-- patient_check_in_value   (one row per answered parameter)
--   A parameter with NO ROW here was not answered. That is deliberate and
--   load-bearing: the engine must distinguish "unknown" from "normal",
--   and a nullable wide column cannot express that difference.
-- ---------------------------------------------------------------------
create table txn.patient_check_in_value (
    check_in_id  uuid    not null references txn.patient_check_in(check_in_id) on delete cascade,
    parameter_id integer not null references meta.parameter(parameter_id)      on delete restrict,
    value_num    numeric(10,2),
    value_bool   boolean,
    value_code   text,
    primary key (check_in_id, parameter_id),
    constraint value_exactly_one check (
        (value_num  is not null)::int
      + (value_bool is not null)::int
      + (value_code is not null)::int = 1)
);

create index check_in_value_by_param on txn.patient_check_in_value (parameter_id);


-- ---------------------------------------------------------------------
-- check_in_evaluation   (what the rules engine decided)
--   check_in_id is nullable on purpose: a missed check-in is evaluated
--   against a synthetic empty reading and still produces a row.
-- ---------------------------------------------------------------------
create table txn.check_in_evaluation (
    evaluation_id       uuid        primary key default gen_random_uuid(),
    check_in_id         uuid        references txn.patient_check_in(check_in_id) on delete cascade,
    episode_id          uuid        not null references txn.patient_episode(episode_id) on delete cascade,
    ruleset_id          integer     not null references meta.ruleset(ruleset_id),
    day_after_discharge smallint    not null,
    tier                text        not null,
    total_score         smallint,
    is_override         boolean     not null default false,
    reasons             jsonb       not null default '[]'::jsonb,
    reason_key          text        not null,      -- hash of sorted reason names, drives dedup
    coverage            numeric(4,3),              -- fraction of expected parameters answered
    evaluated_at        timestamptz not null default now(),

    constraint eval_tier_known
        check (tier in ('normal','review','critical','emergency','missed')),
    constraint eval_coverage_fraction
        check (coverage is null or coverage between 0 and 1),
    constraint eval_reasons_is_array
        check (jsonb_typeof(reasons) = 'array')
);

create index evaluation_timeline
    on txn.check_in_evaluation (episode_id, day_after_discharge, evaluated_at desc);


-- ---------------------------------------------------------------------
-- alert
-- ---------------------------------------------------------------------
create table txn.alert (
    alert_id        uuid        primary key default gen_random_uuid(),
    episode_id      uuid        not null references txn.patient_episode(episode_id) on delete cascade,
    evaluation_id   uuid        not null references txn.check_in_evaluation(evaluation_id),
    tier            text        not null,
    reason_key      text        not null,
    status          text        not null default 'open',
    raised_at       timestamptz not null default now(),
    notified_at     timestamptz,
    acknowledged_at timestamptz,
    acknowledged_by text,
    escalated_at    timestamptz,
    escalated_to    text,
    resolved_at     timestamptz,
    resolved_by     text,
    outcome         text,
    outcome_note    text,

    constraint alert_tier_known
        check (tier in ('review','critical','emergency','missed')),
    constraint alert_status_known
        check (status in ('open','acknowledged','resolved','expired')),
    constraint alert_outcome_known
        check (outcome is null or outcome in
               ('reassured','plan_adjusted','same_day_review','emergency')),
    -- there is no "dismiss": every resolved alert carries an outcome
    constraint alert_resolved_needs_outcome
        check (status <> 'resolved'
               or (outcome is not null and resolved_at is not null)),
    constraint alert_ack_before_resolve
        check (resolved_at is null or acknowledged_at is null
               or resolved_at >= acknowledged_at)
);

-- deduplication: only one live alert per reason set per episode
create unique index alert_one_live_per_reason
    on txn.alert (episode_id, reason_key)
    where status in ('open','acknowledged');

-- the nurse queue
create index alert_queue
    on txn.alert (status, tier, raised_at)
    where status in ('open','acknowledged');

-- the escalation sweep, run every 60s
create index alert_escalation_sweep
    on txn.alert (notified_at)
    where status = 'open' and tier in ('critical','emergency');


-- ---------------------------------------------------------------------
-- alert_notification   (delivery attempts — SMS and push)
-- ---------------------------------------------------------------------
create table txn.alert_notification (
    notification_id uuid        primary key default gen_random_uuid(),
    alert_id        uuid        not null references txn.alert(alert_id) on delete cascade,
    recipient       text        not null,
    channel         text        not null,
    provider_ref    text,
    status          text        not null default 'queued',
    attempts        smallint    not null default 0,
    last_error      text,
    sent_at         timestamptz,
    created_at      timestamptz not null default now(),

    constraint notif_channel_known check (channel in ('push','sms','email')),
    constraint notif_status_known  check (status in ('queued','sent','failed'))
);

create index notification_by_alert on txn.alert_notification (alert_id);

commit;

        ('walk_aid','crutch','Crutches',    2),
        ('walk_aid','cane',  'Walking stick',3),
        ('walk_aid','none',  'No aid',      4)
     ) as v(pcode, code, label, rank) on v.pcode = p.parameter_code;

    (procedure_id, version, review_at, critical_at, status, approved_by, approved_at)

from meta.surgical_procedure where procedure_code = 'TKR';

-- three representative rules: a day-banded numeric, a flat numeric,
-- and a day-gated binary
insert into meta.procedure_parameter
    (ruleset_id, parameter_id, display_order, display_section,
     track_from_day, track_to_day, parameter_weight, scoring_config, question_text)
select r.ruleset_id, p.parameter_id, v.ord, v.section,

join meta.surgical_procedure sp
     on sp.procedure_id = r.procedure_id and sp.procedure_code = 'TKR'

    ('temp_c', 1, 'readings', 0, 42, 3,
     '{"bands":[[0,37.5,0],[37.5,38.0,1],[38.0,38.5,2],[38.5,99,3]]}',

     '{"binary_score":2}',
     'Is fluid still leaking from your wound?')
) as v(pcode, ord, section, from_day, to_day, weight, cfg, question)
  on true
join meta.parameter p on p.parameter_code = v.pcode
where r.version = '1.0.0';

-- =====================================================================
--  Handy view: the nurse queue
-- =====================================================================
create or replace view txn.v_alert_queue as
select a.alert_id,
       a.tier,
       a.status,
       a.raised_at,
       extract(epoch from now() - a.raised_at)::int as age_secs,
       pt.patient_name,
       e.episode_id,
       ev.day_after_discharge,
       ev.reasons,
       h.hospital_name
from txn.alert a
join txn.check_in_evaluation ev on ev.evaluation_id = a.evaluation_id
join txn.patient_episode e      on e.episode_id     = a.episode_id
join txn.patient pt             on pt.patient_id    = e.patient_id
join meta.hospital h            on h.hospital_id    = e.hospital_id
where a.status in ('open','acknowledged')
order by case a.tier when 'emergency' then 1 when 'critical' then 2
                     when 'missed'    then 3 else 4 end,
         a.raised_at;
